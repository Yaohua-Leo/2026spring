\declaretheoremstyle[
    spaceabove=0pt,
    spacebelow=0pt,
    headfont=\bfseries,
    bodyfont=\itshape,
    headpunct={.},
    postheadspace={ }
]{aa-boxed-theorem}

\tcbset{
    aa boxed theorem/.style={
        top=3pt,
        bottom=3pt
    }
}

% 1. Theorem (蓝色风格)
\declaretheorem[name=Theorem, numberwithin=section, style=aa-boxed-theorem]{theorem}
\tcolorboxenvironment{theorem}{
    aa boxed theorem,
    enhanced jigsaw,
    colframe=ThmColor,       % 边框颜色
    colback=ThmBg,           % 背景颜色
    coltitle=white,          
    boxrule=0.5pt,           % 边框粗细
    leftrule=2pt,            % 左边框加粗
    sharp corners=downhill,  % 样式
    breakable,               % 允许跨页
    parbox=false,            % 避免与行间公式组合时出现零高度分段警告
    drop shadow=black!5!white % 轻微阴影
}
% Corollary 使用与 Theorem 相同的风格，但通常紧随其后
\declaretheorem[name=Corollary, numberlike=theorem, style=aa-boxed-theorem]{corollary}
\tcolorboxenvironment{corollary}{
    aa boxed theorem,
    enhanced jigsaw,
    colframe=ThmColor!80!white, 
    colback=ThmBg,
    boxrule=0pt, leftrule=2pt, arc=0pt, outer arc=0pt,
    breakable,
    parbox=false
}
% 2. Definition (绿色风格)
\declaretheorem[name=Definition, numberlike=theorem, style=aa-boxed-theorem]{definition}
\tcolorboxenvironment{definition}{

    enhanced jigsaw,
    colframe=DefColor,
    colback=DefBg,
    boxrule=0.5pt, leftrule=2pt,
    sharp corners=downhill,
    breakable,
    parbox=false,
    drop shadow=black!5!white
}
% 3. Lemma & Proposition (橙色风格)

\tcolorboxenvironment{lemma}{

    colback=LemBg,

\tcolorboxenvironment{proposition}{

    enhanced jigsaw,
    colframe=LemColor,
    colback=LemBg,
    boxrule=0.5pt, leftrule=2pt,
    breakable,
    parbox=false
}
% 4. Example & Remark (简约风格)

    parbox=false
